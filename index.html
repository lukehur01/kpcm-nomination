// ── 공통 헬퍼 ────────────────────────────────────────────── function corsResponse(data) { return ContentService .createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); } // ── GET: 명단 & 자격기준 불러오기 ────────────────────────── function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var action = e.parameter.action || 'candidates'; var type = e.parameter.type; // 'elder' or 'deacon' // 후보자 명단 if (action === 'candidates') { var sheetName = type === 'elder' ? '장로대상자' : '집사대상자'; var sheet = ss.getSheetByName(sheetName); var rows = sheet.getDataRange().getValues(); var groups = {}, order = []; rows.forEach(function(row) { var group = (row[0]||'').toString().trim(); var name = (row[1]||'').toString().trim(); if (!group || !name) return; if (!groups[group]) { groups[group]=[]; order.push(group); } groups[group].push(name); }); return corsResponse({ result:'success', data: order.map(function(g){ return {group:g, names:groups[g]}; }) }); } // 자격 기준 if (action === 'quals') { var sheetName = type === 'elder' ? '장로자격기준' : '집사자격기준'; var sheet = ss.getSheetByName(sheetName); var rows = sheet.getDataRange().getValues(); var quals = []; rows.forEach(function(row) { var ko = (row[1]||'').toString().trim(); var en = (row[2]||'').toString().trim(); if (ko) quals.push({ ko: ko, en: en }); }); return corsResponse({ result:'success', data: quals }); } } catch(err) { return corsResponse({ result:'error', message: err.toString() }); } } // ── POST: 추천서 제출 + 중복 체크 ───────────────────────── function doPost(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); var type = data.type; var sheetName = type === 'elder' ? '장로추천' : '안수집사추천'; var sheet = ss.getSheetByName(sheetName); // ── 헤더 자동 생성 ── if (sheet.getLastRow() === 0) { var headers = ['제출시간','추천인 성명','연락처','날짜']; var max = type === 'elder' ? 3 : 15; for (var i = 1; i <= max; i++) headers.push('추천'+i+'_성명','추천'+i+'_사유'); sheet.appendRow(headers); } // ── 중복 체크 (연락처 기준) ── var phone = data.submitterPhone.replace(/\D/g,''); // 숫자만 추출 var allData = sheet.getDataRange().getValues(); for (var r = 1; r < allData.length; r++) { var existingPhone = (allData[r][2]||'').toString().replace(/\D/g,''); if (existingPhone && existingPhone === phone) { return corsResponse({ result: 'duplicate', message: '이미 추천서를 제출하셨습니다.\nYou have already submitted a nomination form.' }); } } // ── 데이터 저장 ── var row = [new Date(), data.submitterName, data.submitterPhone, data.date]; var maxN = type === 'elder' ? 3 : 15; for (var j = 1; j <= maxN; j++) { var nominee = data.nominees.find(function(n){ return n.num === j; }); row.push(nominee ? nominee.name : '', nominee ? nominee.reason : ''); } sheet.appendRow(row); return corsResponse({ result: 'success' }); } catch(err) { return corsResponse({ result:'error', message: err.toString() }); } }
